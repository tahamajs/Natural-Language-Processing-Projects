@tool
def trip_planning_tool(destination: str, arrival_date: str, departure_date: str, interests: str = ""):
    # Calculate duration
    start = datetime.strptime(arrival_date, "%Y-%m-%d")
    end = datetime.strptime(departure_date, "%Y-%m-%d")
    duration = (end - start).days + 1
    
    # Generate search queries
    queries = [
        f"top tourist attractions in {destination}",
        f"best restaurants local food in {destination}",
        f"cultural activities in {destination}"
    ]
    
    if interests:
        queries.append(f"best {interests} activities in {destination}")
    
    # RAG search
    rag_results = []
    try:
        tbl = db.open_table("book_table")
        for q in queries:
            q_vec = embedding_model.encode([q])[0]
            results = tbl.search(q_vec).limit(2).to_pandas()
            rag_results.extend(results['text'].tolist())
    except:
        pass
    
    # Web search fallback
    web_results = []
    for q in queries:
        response = tavily_client.search(query=q, max_results=2)
        web_results.extend([r['content'] for r in response['results']])
    
    # Compile itinerary
    itinerary = f"Trip itinerary for {destination} ({arrival_date} to {departure_date}):\n\n"
    itinerary += "Tourism Book Information:\n" + "\n".join(rag_results[:5]) + "\n\n"
    itinerary += "Additional Web Sources:\n" + "\n".join(web_results[:5])
    
    return itinerary
\end{lstlisting}

Interactive command-line interface:

def chat_cli():
    print("TravBot CLI - Type 'exit' to quit, 'clear' to reset")
    config = {"configurable": {"thread_id": str(uuid.uuid4())}}
    
    while True:
        user_input = input("\nYou: ").strip()
        if user_input.lower() in ["exit", "quit"]:
            break
        if user_input.lower() == "clear":
            current_chat_history = []
            continue
        
        inputs = {"messages": [HumanMessage(content=user_input)]}
        final_state = app.invoke(inputs, config=config)
        last_message = final_state["messages"][-1]
        
        # Extract tool usage
        tool_calls = set()
        for m in final_state["messages"]:
            if hasattr(m, 'tool_calls') and m.tool_calls:
                for tc in m.tool_calls:
                    tool_calls.add(tc['name'])
        
        if tool_calls:
            print(f"Tools used: {', '.join(tool_calls)}")
        print(f"Bot: {last_message.content}")
