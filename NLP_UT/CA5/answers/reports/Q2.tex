\section{ReAct: Reasoning and Acting in Language Models}

\subsection{Overview of ReAct Paradigm}
The ReAct (Reasoning + Acting) paradigm represents a fundamental advancement in large language model capabilities, enabling synergistic integration of reasoning and action. This approach allows models to interleave verbal reasoning with task-specific actions, creating a more robust and interpretable decision-making process.

\subsection{Core Principles}

\subsubsection{Reasoning-Action Interleaving}
ReAct prompts language models to generate both reasoning traces (thoughts) and actions in an alternating manner:
\begin{itemize}
\item \textbf{Reasoning Phase}: Internal deliberation about task progress, commonsense knowledge, and planning
\item \textbf{Acting Phase}: Interaction with external environments through tool calls or API invocations
\item \textbf{Observation Phase}: Processing feedback from actions to update reasoning
\end{itemize}

\subsubsection{Synergistic Benefits}
\begin{enumerate}
\item \textbf{Enhanced Reasoning}: Actions provide grounding, reducing hallucination
\item \textbf{Improved Planning}: Reasoning enables better action selection and sequencing
\item \textbf{Increased Interpretability}: Thought processes are visible and explainable
\item \textbf{Better Error Recovery}: Dynamic reasoning adapts to unexpected outcomes
\end{enumerate}

        final_state = app.invoke(inputs, config=config)

        # Extract tool usage
        tool_calls = set()
        for m in final_state["messages"]:
            if hasattr(m, 'tool_calls') and m.tool_calls:
                for tc in m.tool_calls:
                    tool_calls.add(tc['name'])
        
        if tool_calls:
            print(f"Tools used: {', '.join(tool_calls)}")
        print(f"Bot: {last_message.content}")
\end{lstlisting}

def download_and_process_airport_data():
    """Download and process IATA airport codes dataset."""
    path = kagglehub.dataset_download("zinovadr/iata-airport-code")
    csv_files = [f for f in os.listdir(path) if f.endswith('.csv')]
    
    airport_df = pd.read_csv(os.path.join(path, csv_files[0]))
    
    # Create city to IATA mapping
    city_to_iata = {}
    for _, row in airport_df.iterrows():
        city = str(row['city']).strip().lower()
        iata = str(row['iata']).strip().upper()
        if city and iata and len(iata) == 3:
            city_to_iata[city] = iata
    
    return city_to_iata

\subsection{LangGraph Agent Implementation}
\begin{lstlisting}[language=Python, caption=Complete LangGraph Agent]
from langgraph import StateGraph, END
from langgraph.prebuilt import ToolNode
from typing import TypedDict, Annotated, Sequence
import operator

# Define state
class AgentState(TypedDict):
    messages: Annotated[Sequence[dict], operator.add]
    context: dict

# Initialize LLM
llm = ChatGoogleGenerativeAI(
    model="gemini-2.0-flash-lite",
    temperature=0.1
)

# Bind tools to LLM
tools = [
    flight_search_tool,
    hotel_search_tool,
    restaurant_search_tool,
    weather_info_tool,
    currency_info_tool,
    faq_search_tool,
    trip_planning_tool
]

llm_with_tools = llm.bind_tools(tools)

# Define nodes
def chatbot_node(state: AgentState) -> AgentState:
    """Process user message and decide on tool usage."""
    messages = state["messages"]
    response = llm_with_tools.invoke(messages)
    
    return {
        "messages": [response],
        "context": state.get("context", {})
    }

def tool_node(state: AgentState) -> AgentState:
    """Execute tool calls."""
    messages = state["messages"]
    last_message = messages[-1]
    
    tool_calls = last_message.tool_calls if hasattr(last_message, 'tool_calls') else []
    
    tool_results = []
    for tool_call in tool_calls:
        tool_name = tool_call["name"]
        tool_args = tool_call["args"]
        
        # Find and execute the tool
        for tool in tools:
            if tool.name == tool_name:
                result = tool.invoke(tool_args)
                tool_results.append({
                    "tool_call_id": tool_call["id"],
                    "content": result
                })
                break
    
    return {
        "messages": tool_results,
        "context": state.get("context", {})
    }

def should_continue(state: AgentState) -> str:
    """Determine whether to continue or end the conversation."""
    messages = state["messages"]
    last_message = messages[-1]
    
    if hasattr(last_message, 'tool_calls') and last_message.tool_calls:
        return "tools"
    else:
        return END

# Create graph
workflow = StateGraph(AgentState)

# Add nodes
workflow.add_node("chatbot", chatbot_node)
workflow.add_node("tools", tool_node)

# Add edges
workflow.add_edge("tools", "chatbot")
workflow.add_conditional_edges("chatbot", should_continue)

# Set entry point
workflow.set_entry_point("chatbot")

# Compile
app = workflow.compile()

# Function to run the agent
def run_travel_agent(query: str) -> str:
    """Run the travel agent with a user query."""
    initial_state = {
        "messages": [{"role": "user", "content": query}],
        "context": {}
    }
    
    result = app.invoke(initial_state)
    
    # Extract final response
    final_message = result["messages"][-1]
    if hasattr(final_message, 'content'):
        return final_message.content
    else:
        return str(final_message)
